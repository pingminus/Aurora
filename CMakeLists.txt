cmake_minimum_required(VERSION 3.24)
project(OpenGod VERSION 0.1.0 LANGUAGES CXX)
if(MSVC)
  # CEF's wrapper uses the static CRT; all linked OpenGod targets must agree.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(OPENGOD_BUILD_BROWSER "Build the Windows CEF browser" OFF)
add_subdirectory(src/core)
if(OPENGOD_BUILD_BROWSER)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The current CEF host requires Windows and MSVC")
  endif()
  add_subdirectory(src/cef)
endif()
