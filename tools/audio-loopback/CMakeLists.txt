cmake_minimum_required(VERSION 3.24)
project(AuroraAudioLoopback LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This verifier requires Windows and the Microsoft Windows SDK")
endif()
add_executable(aurora-audio-loopback main.cpp)
target_compile_features(aurora-audio-loopback PRIVATE cxx_std_20)
target_compile_definitions(aurora-audio-loopback PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(aurora-audio-loopback PRIVATE /W4 /WX /utf-8)
target_link_libraries(aurora-audio-loopback PRIVATE mmdevapi ole32)
include(CTest)
add_test(NAME audio_spectrum_self_test COMMAND aurora-audio-loopback --self-test)
