cmake_minimum_required(VERSION 3.24)
project(AuroraTerminalTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(terminal_tests terminal_test.cpp ../../src/terminal/session.cpp)
target_include_directories(terminal_tests PRIVATE ../../src)
target_compile_definitions(terminal_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(terminal_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(terminal_tests PRIVATE advapi32)
enable_testing()
add_test(NAME conpty_lifecycle COMMAND terminal_tests)
set_tests_properties(conpty_lifecycle PROPERTIES TIMEOUT 40)
